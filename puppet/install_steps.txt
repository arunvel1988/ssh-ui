nano setup-master.sh


#!/bin/bash

# Update system and install required dependencies
echo "Updating system and installing required dependencies..."
sudo apt-get update -y

# Install the Puppet release package
echo "Downloading and installing Puppet release package..."
wget https://apt.puppetlabs.com/puppet8-release-jammy.deb
sudo dpkg -i puppet8-release-jammy.deb
sudo apt-get update -y

# Install Puppet Server
echo "Installing Puppet Server..."
sudo apt-get install puppetserver -y

# Configure JVM memory for Puppet Server
echo "Configuring JVM memory for Puppet Server..."
echo "JAVA_ARGS=\"-Xms2g -Xmx2g\"" | sudo tee -a /etc/default/puppetserver

# Open firewall port for Puppet Server (port 8140)
echo "Allowing firewall port 8140..."
sudo ufw allow 8140

# Start Puppet Server
echo "Starting Puppet Server..."
sudo systemctl start puppetserver

# Enable Puppet Server to start on boot
echo "Enabling Puppet Server to start on boot..."
sudo systemctl enable puppetserver

# Check the status of Puppet Server to ensure it is running
echo "Checking Puppet Server status..."
sudo systemctl status puppetserver

echo "Puppet Master setup completed successfully!"

nano setup-agent.sh


#!/bin/bash

# Update system and install required dependencies
echo "Updating system and installing required dependencies..."
sudo apt-get update -y

# Install Puppet release package
echo "Downloading and installing Puppet release package..."
wget https://apt.puppetlabs.com/puppet8-release-jammy.deb
sudo dpkg -i puppet8-release-jammy.deb
sudo apt-get update -y

# Install Puppet Agent
echo "Installing Puppet Agent..."
sudo apt-get install puppet-agent -y

# Configure Puppet Agent settings
echo "Configuring Puppet Agent..."
sudo mkdir -p /etc/puppetlabs/puppet
echo "[main]" | sudo tee /etc/puppetlabs/puppet/puppet.conf
echo "certname = puppetclient" | sudo tee -a /etc/puppetlabs/puppet/puppet.conf
echo "server = puppetmaster" | sudo tee -a /etc/puppetlabs/puppet/puppet.conf

# Start Puppet Agent
echo "Starting Puppet Agent..."
sudo systemctl start puppet

# Enable Puppet Agent to start on boot
echo "Enabling Puppet Agent to start on boot..."
sudo systemctl enable puppet

# Check the status of Puppet Agent to ensure it is running
echo "Checking Puppet Agent status..."
sudo systemctl status puppet

echo "Puppet Agent setup completed successfully!"
