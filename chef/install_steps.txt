Vagrant.configure("2") do |config|

  config.vm.box = "generic/ubuntu2204"

  # ============================
  # Chef Server
  # ============================
  config.vm.define "chef-server" do |chefserver|
    chefserver.vm.hostname = "chef-server"
    chefserver.vm.network "private_network", ip: "192.168.56.10"

    chefserver.vm.provider "virtualbox" do |vb|
      vb.memory = 4096
      vb.cpus = 2
    end

    chefserver.vm.provision "shell", inline: <<-SHELL
      sudo apt update -y
      sudo apt install curl -y

      curl -L https://packages.chef.io/files/stable/chef-server/latest/ubuntu/24.04/chef-server-core_latest_amd64.deb -o chef-server.deb
      sudo dpkg -i chef-server.deb

      sudo chef-server-ctl reconfigure

      sudo chef-server-ctl user-create admin Arun Admin admin@example.com password --filename /home/vagrant/admin.pem

      sudo chef-server-ctl org-create myorg "My Organization" \
        --association_user admin \
        --filename /home/vagrant/myorg-validator.pem
    SHELL
  end


  # ============================
  # Chef Workstation
  # ============================
  config.vm.define "chef-workstation" do |workstation|
    workstation.vm.hostname = "chef-workstation"
    workstation.vm.network "private_network", ip: "192.168.56.20"

    workstation.vm.provider "virtualbox" do |vb|
      vb.memory = 2048
    end

    workstation.vm.provision "shell", inline: <<-SHELL
      sudo apt update -y
      sudo apt install curl -y

      curl -L https://packages.chef.io/files/stable/chef-workstation/latest/ubuntu/24.04/chef-workstation_latest_amd64.deb -o chef-workstation.deb
      sudo dpkg -i chef-workstation.deb

      mkdir -p /home/vagrant/.chef
      mkdir -p /home/vagrant/cookbooks
    SHELL
  end


  # ============================
  # Node 1
  # ============================
  config.vm.define "node1" do |node1|
    node1.vm.hostname = "node1"
    node1.vm.network "private_network", ip: "192.168.56.30"

    node1.vm.provider "virtualbox" do |vb|
      vb.memory = 1024
    end
  end


  # ============================
  # Node 2
  # ============================
  config.vm.define "node2" do |node2|
    node2.vm.hostname = "node2"
    node2.vm.network "private_network", ip: "192.168.56.40"

    node2.vm.provider "virtualbox" do |vb|
      vb.memory = 1024
    end
  end

end
