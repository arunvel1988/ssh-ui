---
apiVersion: image.openshift.io/v1
kind: ImageStream
metadata:
  name: docker-app-ci
  labels:
    app: docker-app-ci
---

apiVersion: build.openshift.io/v1
kind: BuildConfig
metadata:
  name: docker-app-ci-build
  labels:
    app: docker-app-ci
spec:
  source:
    type: Git
    git:
      uri: https://github.com/arunvel1988/docker_app_ci
      ref: main
    contextDir: .
  strategy:
    type: Docker
    dockerStrategy:
      dockerfilePath: Dockerfile
      forcePull: true
  output:
    to:
      kind: ImageStreamTag
      name: docker-app-ci:latest
  triggers:
    - type: GitHub
      github:
        secret: github-webhook-secret
    - type: ConfigChange
  runPolicy: Serial
---

apiVersion: apps/v1
kind: Deployment
metadata:
  name: docker-app-ci-deploy
  labels:
    app: docker-app-ci
spec:
  replicas: 1
  selector:
    matchLabels:
      app: docker-app-ci
  template:
    metadata:
      labels:
        app: docker-app-ci
    spec:
      containers:
        - name: docker-app-ci
          image: image-registry.openshift-image-registry.svc:5000/arunvel1988-dev/docker-app-ci:latest
          ports:
            - containerPort: 8000

---

apiVersion: v1
kind: Service
metadata:
  name: docker-app-ci-svc
spec:
  selector:
    app: docker-app-ci
  ports:
    - name: http         
      port: 80
      targetPort: 8000
  type: ClusterIP

---
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: docker-app-ci-route
spec:
  to:
    kind: Service
    name: docker-app-ci-svc
  port:
    targetPort: http    

---
